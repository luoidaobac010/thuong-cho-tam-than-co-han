    int count = 0;
    for (int i = 0; i < n - 1; i++) {
        for (int j = i + 1; j < n; j++) {

                count++;

    }

```c
void Hanoi(int n, char A, char B, char C) {
    if (n == 1) 
        transfer(A, C);
    else {
        Hanoi(n - 1, A, C, B);
        transfer(A, C);
        Hanoi(n - 1, B, A, C);
    }
}
```
